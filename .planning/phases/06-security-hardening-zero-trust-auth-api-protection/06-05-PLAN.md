<objective>
Protect the signal ingestion endpoints with API key authentication AND mount a token-bucket rate limit on `POST /v1/signals` (REQ-P6-04 SDK ingest path).

Purpose:
- REQ-P6-03: Close the JWT ExcludedPaths bypass on ingest; SDKs authenticate via `X-Argus-API-Key`.
- REQ-P6-04: The sliding-window Redis limiter from 06-02 covers /auth/login, /auth/refresh, /api/v1/query. The SDK ingest path needs the in-memory `TokenBucket` (internal/resilience/rate_limiter.go) keyed per API-key-prefix for per-client fairness at 10K+ signals/sec.

Output: `/v1/signals`, `/v1/signals/stream`, `/v1/schema/signals` require a valid, scoped API key. `POST /v1/signals` also enforces 10,000 req/s with burst 100 per API key.
</objective>

```go
type APIKey struct {
    ID         uuid.UUID
    UserID     uuid.UUID
    AppID      *uuid.UUID
    Name       string
    KeyPrefix  string
    KeyHash    string
    Scopes     []string
    LastUsedAt *time.Time
    ExpiresAt  *time.Time
    RevokedAt  *time.Time
    CreatedAt  time.Time
}

type ApiKeyStore interface {
    Create(ctx context.Context, k *APIKey) error
    GetByHash(ctx context.Context, hash string) (*APIKey, error)
    ListByUser(ctx context.Context, userID uuid.UUID) ([]*APIKey, error)
    GetByID(ctx context.Context, id uuid.UUID) (*APIKey, error)
    Revoke(ctx context.Context, id uuid.UUID) error
    TouchLastUsed(ctx context.Context, id uuid.UUID, t time.Time) error
}

// PACKAGE-LEVEL — not a method on the store.
// The middleware calls this, passing the RAW key from the header. The function
// hashes internally and does revoked/expired checks + last-used touch.
func ValidateAPIKey(ctx context.Context, store ApiKeyStore, rawKey string) (*APIKey, error)
```

    1. Create `internal/resilience/apikey_token_bucket.go`:

       ```go
       package resilience

       import (
           "fmt"
           "net/http"
           "sync"

           "github.com/argusxdr/argus/internal/auth"  // or whatever the module path is
       )

       // APIKeyTokenBucketLimiter keeps one TokenBucket per api_key_prefix.
       // Keys: "tb:signals:{api_key_prefix}". Burst=100, rate=10000/sec.
       type APIKeyTokenBucketLimiter struct {
           mu      sync.Mutex
           buckets map[string]*TokenBucket
           rate    float64
           burst   float64
       }

       func NewAPIKeyTokenBucketLimiter(rate, burst float64) *APIKeyTokenBucketLimiter {
           return &APIKeyTokenBucketLimiter{
               buckets: make(map[string]*TokenBucket),
               rate:    rate,
               burst:   burst,
           }
       }

       func (l *APIKeyTokenBucketLimiter) bucketFor(prefix string) *TokenBucket {
           l.mu.Lock()
           defer l.mu.Unlock()
           b, ok := l.buckets[prefix]
           if !ok {
               b = NewTokenBucket(l.rate, l.burst)
               l.buckets[prefix] = b
           }
           return b
       }

       // Middleware returns a chi/http middleware that enforces per-api-key limits.
       // Must be mounted AFTER auth.APIKeyMiddleware so the APIKey is in context.
       func (l *APIKeyTokenBucketLimiter) Middleware() func(http.Handler) http.Handler {
           return func(next http.Handler) http.Handler {
               return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                   k, ok := auth.APIKeyFromContext(r.Context())
                   if !ok || k == nil {
                       // No API key in context — should not happen if mounted correctly.
                       // Pass through; upstream middleware enforces authentication.
                       next.ServeHTTP(w, r)
                       return
                   }
                   bucketKey := "tb:signals:" + k.KeyPrefix
                   _ = bucketKey // key name is documented; internal map uses k.KeyPrefix
                   b := l.bucketFor(k.KeyPrefix)
                   if !b.Allow() {
                       w.Header().Set("Retry-After", "1")
                       w.Header().Set("Content-Type", "application/json")
                       w.WriteHeader(http.StatusTooManyRequests)
                       fmt.Fprint(w, `{"error":"signal ingest rate limit exceeded"}`)
                       return
                   }
                   next.ServeHTTP(w, r)
               })
           }
       }
       ```

       Note: the `bucketKey` variable documents the canonical naming convention `tb:signals:{api_key_prefix}` required by the checker. The bucket map is keyed by `k.KeyPrefix` directly (adding the "tb:signals:" prefix to the map key is redundant since this limiter is ingest-signals-specific).
